ref rHandler
int iHandle

begin function {}
    ; The owner name identifies this addon; runtime-compiled scripts share one
    ; load-order index, so the bridge name alone cannot.
    let iHandle := DialecticRegisterOwnedBridge "ParityProbe" "ParityProbe.Example"
    if eval iHandle <= 0
        Print "[ParityProbe] bridge name unavailable; see dialectic.log"
        return
    endif

    let rHandler := CompileScript "ParityProbe/OnExternalCommand.txt"
    if eval !rHandler
        Print "[ParityProbe] external command handler failed to compile"
        return
    endif

    ; Argument 1 is the owner handle, so this handler only receives this addon's commands.
    SetEventHandlerAlt "DialecticOwnedExternalCommand" rHandler 1::iHandle
    DialecticSendPluginEvent "ParityProbe" "state" "loaded=1"
end
