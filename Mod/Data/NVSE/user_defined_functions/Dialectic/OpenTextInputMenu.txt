ref rTarget
string_var sTargetName
string_var sHintTarget
string_var sModeLabel
array_var aModeLabels
float fNarratorMode
float fModeIndex

begin function {}
    WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_textinput_status.tmp" 0 "entered=1\n"
    if MenuMode
        WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_textinput_status.tmp" 0 "entered=1\nblocked=1\nreason=menu_mode\n"
        return
    endif

    ; The mode and narrator flags are runtime state owned by the plugin; they are never
    ; written back to dialectic_custom.ini, so read them from the plugin instead of the INI.
    let fNarratorMode := DialecticGetConfigInt "Narrator" "Mode" 0
    let fModeIndex := DialecticGetConfigInt "Modes" "CurrentIndex" 0
    if eval fNarratorMode > 0
        let fModeIndex := 4
    endif

    ; Index order mirrors ModeLabelFromIndex in Plugin/src/GameLoop.cpp.
    let aModeLabels := ar_List "STANDARD" "WHISPER" "CLOSE" "SHOUT" "NARRATOR" "DIRECTOR" "INJECT EVENT" "INJECT & CHAT" "CHEAT"
    if eval fModeIndex >= 0 && fModeIndex <= 8
        let sModeLabel := aModeLabels[fModeIndex]
    else
        let sModeLabel := sv_Construct "STANDARD"
    endif

    if eval fModeIndex == 4
        let sTargetName := sv_Construct "The Narrator"
    else
        let sHintTarget := ReadStringFromFile "Data\\NVSE\\Plugins\\dialectic_textinput_target.tmp" 1 1
        if eval sHintTarget && (sv_Length sHintTarget) > 0
            let sTargetName := sv_Construct "%z" sHintTarget
        else
            let rTarget := GetCrosshairRef
        endif
        if eval !(sTargetName) && rTarget && IsFormValid rTarget && rTarget.IsActor
            call (CompileScript "Dialectic/CollectActorSnapshot.txt")
            let sTargetName := GetName rTarget
        endif
    endif

    if eval sTargetName && (sv_Length sTargetName) > 0
        WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_textinput_status.tmp" 0 "entered=1 target=%z\n" sTargetName
        ShowTextInputMenu (CompileScript "Dialectic/TextInputSubmit.txt") 600 130 "Talking to %z [%z]" sTargetName sModeLabel
        SetTextInputExtendedProps 0 0 0 32767 2
        call (CompileScript "Dialectic/ApplyTextInputMenuLayout.txt")
    else
        WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_textinput_status.tmp" 0 "entered=1 target=fallback\n"
        ShowTextInputMenu (CompileScript "Dialectic/TextInputSubmit.txt") 600 130 "Talking to target [%z]" sModeLabel
        SetTextInputExtendedProps 0 0 0 32767 2
        call (CompileScript "Dialectic/ApplyTextInputMenuLayout.txt")
    endif

    if eval sTargetName
        sv_Destruct sTargetName
    endif
    if eval sHintTarget
        sv_Destruct sHintTarget
    endif
    if eval sModeLabel
        sv_Destruct sModeLabel
    endif
end
