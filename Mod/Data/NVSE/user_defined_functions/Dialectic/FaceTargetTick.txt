string_var sSpeakerRef
string_var sTargetRef
string_var sYaw

float fPrid
float fFaceObject
float fSetAngle

begin function {}
    if eval (FileExists "NVSE\Plugins\dialectic_facing_speaker_ref.txt") && (FileExists "NVSE\Plugins\dialectic_facing_target_ref.txt")
        let sSpeakerRef := ReadStringFromFile "Data\NVSE\Plugins\dialectic_facing_speaker_ref.txt" 1 1
        let sTargetRef := ReadStringFromFile "Data\NVSE\Plugins\dialectic_facing_target_ref.txt" 1 1
        let sYaw := ReadStringFromFile "Data\NVSE\Plugins\dialectic_facing_yaw.txt" 1 1

        if sSpeakerRef && sv_Length sSpeakerRef > 0 && sTargetRef && sv_Length sTargetRef > 0
            set fPrid to Console "prid %z" sSpeakerRef
            if sYaw && sv_Length sYaw > 0
                set fSetAngle to Console "SetAngle Z %z" sYaw
            endif
            set fFaceObject to Console "FaceObject %z" sTargetRef
            WriteStringToFile "Data\NVSE\Plugins\dialectic_facing_status.txt" 0 "active=0 one_shot=1 speaker=%z target=%z prid=%.0f setangle=%.0f faceobject=%.0f\n" sSpeakerRef sTargetRef fPrid fSetAngle fFaceObject

            WriteStringToFile "Data\NVSE\Plugins\dialectic_facing_speaker_ref.txt" 0 "%e"
            WriteStringToFile "Data\NVSE\Plugins\dialectic_facing_target_ref.txt" 0 "%e"
            WriteStringToFile "Data\NVSE\Plugins\dialectic_facing_yaw.txt" 0 "%e"
        endif
    endif

    if sSpeakerRef
        sv_Destruct sSpeakerRef
    endif
    if sTargetRef
        sv_Destruct sTargetRef
    endif
    if sYaw
        sv_Destruct sYaw
    endif

    CallAfterSeconds 0.05 (CompileScript "Dialectic/FaceTargetTick.txt") 0
end
