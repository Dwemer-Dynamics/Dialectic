string_var sRequestId
string_var sSpeakerRef
string_var sTargetRef
string_var sSpeakerMod
string_var sSpeakerLocal
string_var sTargetMod
string_var sTargetLocal
string_var sSpeakerTeamBefore
string_var sTargetTeamBefore
string_var sSpeakerAggBefore
string_var sSpeakerConfBefore
string_var sSpeakerAssistBefore
string_var sTargetAggBefore
string_var sTargetConfBefore
string_var sTargetAssistBefore

ref rSpeaker
ref rTarget

float fSpeakerTeamBefore
float fTargetTeamBefore
float fSpeakerAggBefore
float fSpeakerConfBefore
float fSpeakerAssistBefore
float fTargetAggBefore
float fTargetConfBefore
float fTargetAssistBefore
float fSpeakerCombat
float fTargetCombat

int iSpeakerMod
int iSpeakerLocal
int iTargetMod
int iTargetLocal

begin function {}
    if eval !(FileExists "NVSE\\Plugins\\dialectic_attack_cleanup.tmp")
        return
    endif

    let sRequestId := ReadStringFromFile "Data\\NVSE\\Plugins\\dialectic_attack_cleanup.tmp" 1 1
    let sSpeakerRef := ReadStringFromFile "Data\\NVSE\\Plugins\\dialectic_attack_cleanup.tmp" 2 1
    let sTargetRef := ReadStringFromFile "Data\\NVSE\\Plugins\\dialectic_attack_cleanup.tmp" 3 1
    let sSpeakerMod := ReadStringFromFile "Data\\NVSE\\Plugins\\dialectic_attack_cleanup.tmp" 4 1
    let sSpeakerLocal := ReadStringFromFile "Data\\NVSE\\Plugins\\dialectic_attack_cleanup.tmp" 5 1
    let sTargetMod := ReadStringFromFile "Data\\NVSE\\Plugins\\dialectic_attack_cleanup.tmp" 6 1
    let sTargetLocal := ReadStringFromFile "Data\\NVSE\\Plugins\\dialectic_attack_cleanup.tmp" 7 1
    let sSpeakerTeamBefore := ReadStringFromFile "Data\\NVSE\\Plugins\\dialectic_attack_cleanup.tmp" 8 1
    let sTargetTeamBefore := ReadStringFromFile "Data\\NVSE\\Plugins\\dialectic_attack_cleanup.tmp" 9 1
    let sSpeakerAggBefore := ReadStringFromFile "Data\\NVSE\\Plugins\\dialectic_attack_cleanup.tmp" 10 1
    let sSpeakerConfBefore := ReadStringFromFile "Data\\NVSE\\Plugins\\dialectic_attack_cleanup.tmp" 11 1
    let sSpeakerAssistBefore := ReadStringFromFile "Data\\NVSE\\Plugins\\dialectic_attack_cleanup.tmp" 12 1
    let sTargetAggBefore := ReadStringFromFile "Data\\NVSE\\Plugins\\dialectic_attack_cleanup.tmp" 13 1
    let sTargetConfBefore := ReadStringFromFile "Data\\NVSE\\Plugins\\dialectic_attack_cleanup.tmp" 14 1
    let sTargetAssistBefore := ReadStringFromFile "Data\\NVSE\\Plugins\\dialectic_attack_cleanup.tmp" 15 1

    if eval !(sRequestId) || sv_Length sRequestId <= 0
        WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_attack_cleanup.tmp" 0 "%e"
        if sRequestId
            sv_Destruct sRequestId
        endif
        if sSpeakerRef
            sv_Destruct sSpeakerRef
        endif
        if sTargetRef
            sv_Destruct sTargetRef
        endif
        if sSpeakerMod
            sv_Destruct sSpeakerMod
        endif
        if sSpeakerLocal
            sv_Destruct sSpeakerLocal
        endif
        if sTargetMod
            sv_Destruct sTargetMod
        endif
        if sTargetLocal
            sv_Destruct sTargetLocal
        endif
        if sSpeakerTeamBefore
            sv_Destruct sSpeakerTeamBefore
        endif
        if sTargetTeamBefore
            sv_Destruct sTargetTeamBefore
        endif
        if sSpeakerAggBefore
            sv_Destruct sSpeakerAggBefore
        endif
        if sSpeakerConfBefore
            sv_Destruct sSpeakerConfBefore
        endif
        if sSpeakerAssistBefore
            sv_Destruct sSpeakerAssistBefore
        endif
        if sTargetAggBefore
            sv_Destruct sTargetAggBefore
        endif
        if sTargetConfBefore
            sv_Destruct sTargetConfBefore
        endif
        if sTargetAssistBefore
            sv_Destruct sTargetAssistBefore
        endif
        return
    endif

    let iSpeakerMod := ToNumber sSpeakerMod
    let iSpeakerLocal := ToNumber sSpeakerLocal
    let iTargetMod := ToNumber sTargetMod
    let iTargetLocal := ToNumber sTargetLocal
    let fSpeakerTeamBefore := ToNumber sSpeakerTeamBefore
    let fTargetTeamBefore := ToNumber sTargetTeamBefore
    let fSpeakerAggBefore := ToNumber sSpeakerAggBefore
    let fSpeakerConfBefore := ToNumber sSpeakerConfBefore
    let fSpeakerAssistBefore := ToNumber sSpeakerAssistBefore
    let fTargetAggBefore := ToNumber sTargetAggBefore
    let fTargetConfBefore := ToNumber sTargetConfBefore
    let fTargetAssistBefore := ToNumber sTargetAssistBefore

    let rSpeaker := 0
    let rTarget := 0

    if eval iSpeakerLocal > 0
        let rSpeaker := BuildRef iSpeakerMod iSpeakerLocal
        if eval rSpeaker && !(rSpeaker.IsActor)
            let rSpeaker := 0
        endif
    endif

    if eval iTargetLocal > 0
        let rTarget := BuildRef iTargetMod iTargetLocal
        if eval rTarget && !(rTarget.IsActor)
            let rTarget := 0
        endif
    endif

    if eval sTargetRef && (sv_Compare sTargetRef "00000014") == 0
        let rTarget := PlayerRef
    endif

    if eval !(rSpeaker)
        WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_attack_cleanup_status.txt" 0 "id=%z speaker=%z target_ref=%z error=speaker_ref_not_loaded%r" sRequestId sSpeakerRef sTargetRef
        WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_attack_cleanup.tmp" 0 "%e"
        if sRequestId
            sv_Destruct sRequestId
        endif
        if sSpeakerRef
            sv_Destruct sSpeakerRef
        endif
        if sTargetRef
            sv_Destruct sTargetRef
        endif
        if sSpeakerMod
            sv_Destruct sSpeakerMod
        endif
        if sSpeakerLocal
            sv_Destruct sSpeakerLocal
        endif
        if sTargetMod
            sv_Destruct sTargetMod
        endif
        if sTargetLocal
            sv_Destruct sTargetLocal
        endif
        if sSpeakerTeamBefore
            sv_Destruct sSpeakerTeamBefore
        endif
        if sTargetTeamBefore
            sv_Destruct sTargetTeamBefore
        endif
        if sSpeakerAggBefore
            sv_Destruct sSpeakerAggBefore
        endif
        if sSpeakerConfBefore
            sv_Destruct sSpeakerConfBefore
        endif
        if sSpeakerAssistBefore
            sv_Destruct sSpeakerAssistBefore
        endif
        if sTargetAggBefore
            sv_Destruct sTargetAggBefore
        endif
        if sTargetConfBefore
            sv_Destruct sTargetConfBefore
        endif
        if sTargetAssistBefore
            sv_Destruct sTargetAssistBefore
        endif
        return
    endif

    let fSpeakerCombat := rSpeaker.IsInCombat
    let fTargetCombat := 0
    if eval rTarget && rTarget != PlayerRef
        let fTargetCombat := rTarget.IsInCombat
    endif

    if eval fSpeakerCombat > 0 || fTargetCombat > 0
        WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_attack_cleanup_status.txt" 0 "id=%z speaker=%z target_ref=%z cleanup=pending combat=1%r" sRequestId sSpeakerRef sTargetRef
        CallAfterSeconds 5.00 (CompileScript "Dialectic/AttackCleanupTick.txt") 0
        if sRequestId
            sv_Destruct sRequestId
        endif
        if sSpeakerRef
            sv_Destruct sSpeakerRef
        endif
        if sTargetRef
            sv_Destruct sTargetRef
        endif
        if sSpeakerMod
            sv_Destruct sSpeakerMod
        endif
        if sSpeakerLocal
            sv_Destruct sSpeakerLocal
        endif
        if sTargetMod
            sv_Destruct sTargetMod
        endif
        if sTargetLocal
            sv_Destruct sTargetLocal
        endif
        if sSpeakerTeamBefore
            sv_Destruct sSpeakerTeamBefore
        endif
        if sTargetTeamBefore
            sv_Destruct sTargetTeamBefore
        endif
        if sSpeakerAggBefore
            sv_Destruct sSpeakerAggBefore
        endif
        if sSpeakerConfBefore
            sv_Destruct sSpeakerConfBefore
        endif
        if sSpeakerAssistBefore
            sv_Destruct sSpeakerAssistBefore
        endif
        if sTargetAggBefore
            sv_Destruct sTargetAggBefore
        endif
        if sTargetConfBefore
            sv_Destruct sTargetConfBefore
        endif
        if sTargetAssistBefore
            sv_Destruct sTargetAssistBefore
        endif
        return
    endif

    rSpeaker.ForceActorDetectionValue 0
    rSpeaker.RemoveScriptPackage
    rSpeaker.RemoveFromFaction DialecticAttackFaction
    rSpeaker.SetAV Aggression fSpeakerAggBefore
    rSpeaker.SetAV Confidence fSpeakerConfBefore
    rSpeaker.SetAV Assistance fSpeakerAssistBefore
    if eval fSpeakerTeamBefore > 0
        rSpeaker.SetPlayerTeammate 1
    else
        rSpeaker.SetPlayerTeammate 0
    endif
    rSpeaker.EvaluatePackage

    if eval rTarget && rTarget != PlayerRef
        rTarget.ForceActorDetectionValue 0
        rTarget.RemoveFromFaction DialecticWaitFaction
        rTarget.SetAV Aggression fTargetAggBefore
        rTarget.SetAV Confidence fTargetConfBefore
        rTarget.SetAV Assistance fTargetAssistBefore
        if eval fTargetTeamBefore > 0
            rTarget.SetPlayerTeammate 1
        else
            rTarget.SetPlayerTeammate 0
        endif
        rTarget.EvaluatePackage
    endif

    WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_attack_cleanup_status.txt" 0 "id=%z speaker=%z target_ref=%z cleanup=1 restored=1%r" sRequestId sSpeakerRef sTargetRef
    WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_attack_cleanup.tmp" 0 "%e"

    if sRequestId
        sv_Destruct sRequestId
    endif
    if sSpeakerRef
        sv_Destruct sSpeakerRef
    endif
    if sTargetRef
        sv_Destruct sTargetRef
    endif
    if sSpeakerMod
        sv_Destruct sSpeakerMod
    endif
    if sSpeakerLocal
        sv_Destruct sSpeakerLocal
    endif
    if sTargetMod
        sv_Destruct sTargetMod
    endif
    if sTargetLocal
        sv_Destruct sTargetLocal
    endif
    if sSpeakerTeamBefore
        sv_Destruct sSpeakerTeamBefore
    endif
    if sTargetTeamBefore
        sv_Destruct sTargetTeamBefore
    endif
    if sSpeakerAggBefore
        sv_Destruct sSpeakerAggBefore
    endif
    if sSpeakerConfBefore
        sv_Destruct sSpeakerConfBefore
    endif
    if sSpeakerAssistBefore
        sv_Destruct sSpeakerAssistBefore
    endif
    if sTargetAggBefore
        sv_Destruct sTargetAggBefore
    endif
    if sTargetConfBefore
        sv_Destruct sTargetConfBefore
    endif
    if sTargetAssistBefore
        sv_Destruct sTargetAssistBefore
    endif
end
