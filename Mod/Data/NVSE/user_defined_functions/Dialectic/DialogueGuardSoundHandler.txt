ref rSoundForm

string_var sFilePath
string_var sBlockedPath

int iFlags
float fStopped
int iMatched

begin Function {sFilePath, iFlags, rSoundForm}
    if eval !(FileExists "NVSE\\Plugins\\dialectic_dialogue_guard_blocked_voice.txt")
        return
    endif

    let sBlockedPath := ReadStringFromFile "Data\\NVSE\\Plugins\\dialectic_dialogue_guard_blocked_voice.txt" 1 1
    if eval !(sBlockedPath) || (sv_Length sBlockedPath) <= 0
        if sBlockedPath
            sv_Destruct sBlockedPath
        endif
        return
    endif

    if eval !(sFilePath) || (sv_Length sFilePath) <= 0
        sv_Destruct sBlockedPath
        return
    endif

    let iMatched := 0
    if eval (sv_Compare sFilePath sBlockedPath) == 0
        let iMatched := 1
    elseif eval (sv_Find sBlockedPath sFilePath) > -1
        let iMatched := 1
    elseif eval (sv_Find sFilePath sBlockedPath) > -1
        let iMatched := 1
    endif

    if eval iMatched == 1
        if eval rSoundForm && IsFormValid rSoundForm
            StopSound rSoundForm
            let fStopped := 1
        else
            let fStopped := 0
        endif
        WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_dialogue_guard_sound_status.txt" 0 "matched=1 stopped=%g flags=%g%r" fStopped iFlags
        WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_dialogue_guard_sound_status.txt" 1 "path=%z%r" sFilePath
        WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_dialogue_guard_sound_status.txt" 1 "blocked_path=%z%r" sBlockedPath
        WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_dialogue_guard_blocked_voice.txt" 0 "%e"
    endif

    sv_Destruct sBlockedPath
end
