int iMenuMode
int iTradeMenuMode
int iPipboyOpen
int iPauseMenuOpen
int iDialogueMenuOpen
int iCombat

begin function {}
    CallAfterSeconds 0.05 (CompileScript "Dialectic/GameStateTick.txt") 1

    if FileExists "Data\\NVSE\\Plugins\\dialectic_actor_snapshot_request.tmp"
        call (CompileScript "Dialectic/CollectActorSnapshot.txt")
    elseif FileExists "NVSE\\Plugins\\dialectic_actor_snapshot_request.tmp"
        call (CompileScript "Dialectic/CollectActorSnapshot.txt")
    endif

    let iMenuMode := 0
    if MenuMode
        let iMenuMode := 1
    endif
    let iTradeMenuMode := 0
    if MenuMode 1008
        let iTradeMenuMode := 1
    endif
    if MenuMode 1053
        let iTradeMenuMode := 1
    endif
    if MenuMode 1016
        let iTradeMenuMode := 1
    endif
    let iPipboyOpen := 0
    if MenuMode 1
        let iPipboyOpen := 1
    endif
    if MenuMode 1023
        let iPipboyOpen := 1
    endif
    let iPauseMenuOpen := 0
    if MenuMode 1013
        let iPauseMenuOpen := 1
    endif
    let iDialogueMenuOpen := 0
    if MenuMode 1009
        let iDialogueMenuOpen := 1
    endif
    let iCombat := PlayerRef.IsInCombat
    let iMenuMode := iTradeMenuMode
    if iPipboyOpen
        let iMenuMode := 1
    endif
    if iPauseMenuOpen
        let iMenuMode := 1
    endif

    WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_game_state.tmp" 0 "source=game_state%r"
    WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_game_state.tmp" 1 "menu_mode=%g%r" iMenuMode
    WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_game_state.tmp" 1 "trade_menu_mode=%g%r" iTradeMenuMode
    WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_game_state.tmp" 1 "pipboy_open=%g%r" iPipboyOpen
    WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_game_state.tmp" 1 "pause_menu_open=%g%r" iPauseMenuOpen
    WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_game_state.tmp" 1 "dialogue_menu_open=%g%r" iDialogueMenuOpen
    WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_game_state.tmp" 1 "paused=%g%r" iMenuMode
    WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_game_state.tmp" 1 "combat=%g%r" iCombat

    if iTradeMenuMode
        WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_trade_status.txt" 0 "status=waiting_trade_menu%r"
    endif
end
