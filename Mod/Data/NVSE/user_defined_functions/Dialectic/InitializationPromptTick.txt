float fPromptVersion

Begin Function {}
    set fPromptVersion to GetINIFloat "Setup:InitializationPromptVersion" "Data\\NVSE\\Plugins\\dialectic_custom.ini"
    if eval fPromptVersion >= 1
        return
    endif

    if MenuMode
        CallAfterSeconds 2.0 (CompileScript "Dialectic/InitializationPromptTick.txt") 0
        return
    endif

    SetINIIntC "Setup:InitializationPromptVersion" 1 "Data\\NVSE\\Plugins\\dialectic_custom.ini"
    MessageBoxExAlt (CompileScript "Dialectic/InitializationPromptSelect.gek") "^Initialize Dialectic^Send locations, factions, and voices to the server?|Initialize Now|Not Now"
End
