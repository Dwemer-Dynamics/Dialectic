string_var sLastHaltRequestId
string_var sRequestId
string_var sActorRef
string_var sActorMod
string_var sActorLocal
string_var sHaltIcon
string_var sHaltMessage

ref rActor

int iLine
int iModLine
int iLocalLine
int iActorMod
int iActorLocal
float fResult1
float fResult2
float fResult3
float fResult4
float fResult5

begin function {}
    if eval !(FileExists "NVSE\\Plugins\\dialectic_halt_actions.tmp")
        CallAfterSeconds 0.25 (CompileScript "Dialectic/HaltActionsTick.txt") 0
        return
    endif

    let sRequestId := ReadStringFromFile "Data\\NVSE\\Plugins\\dialectic_halt_actions.tmp" 1 1
    if eval !(sRequestId) || sv_Length sRequestId <= 0
        WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_halt_actions.tmp" 0 "%e"
        CallAfterSeconds 0.25 (CompileScript "Dialectic/HaltActionsTick.txt") 0
        return
    endif

    if eval sLastHaltRequestId && (sv_Compare sRequestId sLastHaltRequestId) == 0
        WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_halt_actions.tmp" 0 "%e"
        CallAfterSeconds 0.25 (CompileScript "Dialectic/HaltActionsTick.txt") 0
        return
    endif

    if sLastHaltRequestId
        sv_Destruct sLastHaltRequestId
    endif
    let sLastHaltRequestId := sv_Construct "%z" sRequestId
    let sHaltIcon := sv_Construct "#5"
    let sHaltMessage := sv_Construct "[DIALECTIC] Halting AI actions"
    MessageExAlt 4 "%z|%z" sHaltIcon sHaltMessage
    WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_halt_actions_status.txt" 0 "id=%z status=started%r" sRequestId
    sv_Destruct sHaltIcon
    sv_Destruct sHaltMessage

    let iLine := 2
    while iLine < 194
        let sActorRef := ReadStringFromFile "Data\\NVSE\\Plugins\\dialectic_halt_actions.tmp" iLine 1
        if eval !(sActorRef) || sv_Length sActorRef <= 0
            let iLine := 194
        else
            let iModLine := iLine + 1
            let iLocalLine := iLine + 2
            let sActorMod := ReadStringFromFile "Data\\NVSE\\Plugins\\dialectic_halt_actions.tmp" iModLine 1
            let sActorLocal := ReadStringFromFile "Data\\NVSE\\Plugins\\dialectic_halt_actions.tmp" iLocalLine 1
            let iActorMod := ToNumber sActorMod
            let iActorLocal := ToNumber sActorLocal
            let rActor := 0

            if eval iActorLocal > 0
                let rActor := BuildRef iActorMod iActorLocal
                if eval rActor && !(rActor.IsActor)
                    let rActor := 0
                endif
            endif

            set fResult1 to Console "prid %z" sActorRef
            set fResult5 to Console "SetActorsAI 1"

            if eval rActor
                rActor.SetRestrained 0
                rActor.StopCombat
                rActor.SetWeaponOut 0
                rActor.ForceActorDetectionValue 0
                rActor.RemoveScriptPackage
                rActor.RemoveFromFaction DialecticMoveToFaction
                rActor.RemoveFromFaction DialecticFollowFaction
                rActor.RemoveFromFaction DialecticTravelFaction
                rActor.RemoveFromFaction DialecticWaitFaction
                rActor.RemoveFromFaction DialecticSeatFaction
                rActor.RemoveFromFaction DialecticAttackFaction
                rActor.EvaluatePackage
                let fResult1 := 1
                let fResult2 := 1
                let fResult3 := 1
                let fResult4 := 1
                WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_halt_actions_status.txt" 1 "id=%z actor=%z direct_ref=1 actors_ai=%.0f stopcombat=%.0f sheathe=%.0f evaluate=%.0f%r" sRequestId sActorRef fResult5 fResult2 fResult3 fResult4
            else
                WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_halt_actions_status.txt" 1 "id=%z actor=%z actors_ai=%.0f error=actor_ref_not_loaded%r" sRequestId sActorRef fResult5
            endif
        endif

        if sActorRef
            sv_Destruct sActorRef
        endif
        if sActorMod
            sv_Destruct sActorMod
        endif
        if sActorLocal
            sv_Destruct sActorLocal
        endif
        let iLine += 3
    loop

    WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_halt_actions.tmp" 0 "%e"
    WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_attack_state.tmp" 0 "%e"
    WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_attack_cleanup.tmp" 0 "%e"
    WriteStringToFile "Data\\NVSE\\Plugins\\dialectic_follow_state.tmp" 0 "%e"

    if sRequestId
        sv_Destruct sRequestId
    endif

    CallAfterSeconds 0.25 (CompileScript "Dialectic/HaltActionsTick.txt") 0
end
